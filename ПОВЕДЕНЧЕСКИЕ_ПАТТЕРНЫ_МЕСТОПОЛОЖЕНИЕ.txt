#if DEBUG
    let examples = BehavioralPatternsExamples()
    examples.runAllExamples()
#endif

Или запустите отдельные примеры:

let examples = BehavioralPatternsExamples()
examples.strategyExample1_SortByName()        // Strategy
examples.mementoExample1_SaveAndRestore()     // Memento
examples.templateMethodExample()              // Template Method
examples.commandExample()                     // Command
